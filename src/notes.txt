// ✅ Replace Nominatim with Photon — faster, no key, autocomplete-ready
const searchPlace = async (placeName) => {
  const url = `https://photon.komoot.io/api/?q=${encodeURIComponent(placeName)}&limit=5&lang=en`
  const res  = await fetch(url)
  const data = await res.json()

  // Returns GeoJSON FeatureCollection
  return data.features.map(f => ({
    name:    f.properties.name,
    city:    f.properties.city,
    country: f.properties.country,
    lat:     f.geometry.coordinates[1],  // note: GeoJSON is [lng, lat]
    lng:     f.geometry.coordinates[0],
  }))
}

// Usage
const results = await searchPlace("Eiffel Tower")
// → [{ name: "Eiffel Tower", city: "Paris", lat: 48.8584, lng: 2.2945 }]


const GH_KEY = "YOUR_GRAPHHOPPER_API_KEY"

const getRoute = async (source, dest) => {
  const url = `https://graphhopper.com/api/1/route?` +
    `point=${source.lat},${source.lng}` +
    `&point=${dest.lat},${dest.lng}` +
    `&profile=foot` +           // foot | car | bike
    `&instructions=true` +      // ← gives turn-by-turn steps
    `&points_encoded=false` +   // ← returns plain [lng,lat] arrays, not encoded polyline
    `&locale=en` +
    `&key=${GH_KEY}`

  const res  = await fetch(url)
  const data = await res.json()
  const path = data.paths[0]

  return {
    // Full route geometry as [lng, lat] pairs
    coordinates: path.points.coordinates,

    distanceMeters: path.distance,
    durationMs:     path.time,

    // ✅ THIS is what OSRM was missing — structured turn instructions
    steps: path.instructions.map(step => ({
      text:           step.text,            // "Turn right onto Main St"
      distanceMeters: step.distance,
      durationMs:     step.time,
      sign:           step.sign,            // see sign codes below 🔑
      exitNumber:     step.exit_number,     // for roundabouts
      streetName:     step.street_name,
      // coordinate index into path.points where this step starts
      pointIndex:     step.interval[0],
      coords:         path.points.coordinates[step.interval[0]], // [lng, lat]
    }))
  }
}
